// ─── CJENIK DATABASE ───────────────────────────────────────────────
const CJENIK = ''' + compact + ''';

document.getElementById('dbCount').textContent = CJENIK.length.toLocaleString('hr') + ' stavki u bazi';

// ─── NORMALIZATION ─────────────────────────────────────────────────
const DIAKRITICI = {č:'c',ć:'c',š:'s',ž:'z',đ:'d',Č:'c',Ć:'c',Š:'s',Ž:'z',Đ:'d'};
function norm(s) {
  if (!s) return '';
  return s.toLowerCase()
    .replace(/[čćšžđČĆŠŽĐ]/g, c => DIAKRITICI[c]||c)
    .replace(/[^a-z0-9\\s]/g,' ')
    .replace(/\\s+/g,' ').trim();
}

// ─── DIMENSION EXTRACTION ──────────────────────────────────────────
function extractDims(s) {
  const dims = [];
  const re = /(\\d+(?:[.,]\\d+)?)\\s*(mm|cm|m(?!³|²|3|2)|m²|m³|m2|m3|kg|t(?!\\w))/gi;
  let m;
  while((m=re.exec(s))!==null) {
    dims.push({v:parseFloat(m[1].replace(',','.')), u:m[2].toLowerCase()});
  }
  return dims;
}

function dimScore(d1,d2) {
  if(!d1.length||!d2.length) return 0.5;
  let hits=0,total=0;
  for(const a of d1) for(const b of d2) {
    if(a.u===b.u){total++;if(Math.abs(a.v-b.v)/Math.max(a.v,0.001)<0.06)hits++;}
  }
  return total===0?0.5:hits/total;
}

// ─── JACCARD ON TOKENS ─────────────────────────────────────────────
const STOP = new Set(['i','u','od','do','na','za','s','se','je','mm','cm','sa','iz']);
function tokenize(s){return new Set(norm(s).split(' ').filter(t=>t.length>1&&!STOP.has(t)));}
function jaccard(a,b){const u=new Set([...a,...b]);if(!u.size)return 0;return [...a].filter(x=>b.has(x)).length/u.size;}

// ─── LEVENSHTEIN ───────────────────────────────────────────────────
function lev(a,b){
  const dp=[];
  for(let i=0;i<=a.length;i++){dp[i]=[i];for(let j=1;j<=b.length;j++)dp[i][j]=i===0?j:0;}
  for(let i=1;i<=a.length;i++) for(let j=1;j<=b.length;j++)
    dp[i][j]=a[i-1]===b[j-1]?dp[i-1][j-1]:1+Math.min(dp[i-1][j],dp[i][j-1],dp[i-1][j-1]);
  return dp[a.length][b.length];
}
function levSim(a,b){const mx=Math.max(a.length,b.length);return mx?1-lev(a,b)/mx:1;}

// ─── MAIN MATCH ENGINE ─────────────────────────────────────────────
function match(query, topN=5) {
  const qNorm = norm(query);
  const qTok  = tokenize(query);
  const qDim  = extractDims(query);

  return CJENIK
    .map(item => {
      const iNorm = norm(item.naziv);
      const iTok  = tokenize(item.naziv);
      const iDim  = extractDims(item.naziv);

      const jScore = jaccard(qTok, iTok);
      const lScore = levSim(qNorm.slice(0,60), iNorm.slice(0,60));
      const dScore = dimScore(qDim, iDim);

      // Fuzzy: check if query words appear in item name
      const qWords = [...qTok].filter(w=>w.length>2);
      const fuzzy  = qWords.length ? qWords.filter(w=>iNorm.includes(w)).length/qWords.length : 0;

      const raw = fuzzy*0.35 + jScore*0.30 + lScore*0.15 + dScore*0.20;

      // Dimension penalty: if query has dims and item has dims that don't match → penalize
      let mul = 1.0;
      if(qDim.length && iDim.length && dScore===0) mul=0.6;
      if(qDim.length && iDim.length && dScore===1) mul=1.2;

      return {...item, confidence:Math.min(100,Math.round(raw*mul*100))};
    })
    .filter(x=>x.confidence>=20)
    .sort((a,b)=>b.confidence-a.confidence)
    .slice(0,topN);
}

// ─── STATE ─────────────────────────────────────────────────────────
let queue = [];      // {id, code, name, status:'pending'|'done'|'skip', match, allMatches, selectedIdx}
let currentIdx = -1;
let altsOpen = false;

function generateId(){return 'ST'+Date.now().toString(36).toUpperCase();}

function addItem(){
  const input = document.getElementById('newItemInput');
  const val = input.value.trim();
  if(!val) return;

  const matches = match(val, 8);
  const item = {
    id: generateId(),
    code: 'ST-' + (queue.length+1).toString().padStart(3,'0'),
    name: val,
    status: 'pending',
    allMatches: matches,
    selectedIdx: 0
  };
  queue.push(item);
  input.value = '';

  renderQueue();
  setActive(queue.length-1);
  updateStats();
  showToast('Stavka dodana — ' + matches.length + ' prijedloga pronađeno');
  document.getElementById('emptyState').style.display='none';
  document.getElementById('reconcileView').style.display='block';
}

function setActive(idx){
  currentIdx = idx;
  altsOpen = false;
  renderQueue();
  renderMatch();
  renderProgress();
}

// ─── RENDER QUEUE ──────────────────────────────────────────────────
function renderQueue(){
  const el = document.getElementById('queueList');
  document.getElementById('queueCount').textContent = '('+queue.length+')';
  if(!queue.length){el.innerHTML='<div class="queue-empty">Dodajte stavke iz troškovnika radi usporedbe s cjenikom.</div>';return;}
  el.innerHTML = queue.map((item,i)=>{
    const best = item.allMatches[item.selectedIdx||0];
    const conf = best?best.confidence:0;
    const statusLabel = item.status==='done'?'Prihvaćeno':item.status==='skip'?'Preskočeno':'Na čekanju';
    return `<div class="queue-item ${i===currentIdx?'active':''} ${item.status==='done'||item.status==='skip'?'done':''}" onclick="setActive(${i})">
      <div class="qi-code">${item.code}</div>
      <div class="qi-name">${item.name.length>60?item.name.slice(0,60)+'…':item.name}</div>
      <div class="qi-meta">
        <span class="qi-status ${item.status}">${statusLabel}</span>
        ${best?`<span class="qi-conf">${conf}% sigurnost</span>`:''}
      </div>
    </div>`;
  }).join('');
}

function confLabel(c){return c>=75?'Visoka sigurnost':c>=50?'Srednja sigurnost':'Niska sigurnost';}
function confBg(c){return c>=75?'var(--green-bg)':c>=50?'var(--amber-bg)':'var(--red-bg)';}
function scoreClass(c){return c>=75?'style="background:var(--green-bg);color:var(--green-text)"':c>=50?'style="background:var(--amber-bg);color:var(--amber-text)"':'style="background:var(--red-bg);color:var(--red-text)"';}

function renderMatch(){
  if(currentIdx<0||currentIdx>=queue.length) return;
  const item = queue[currentIdx];

  // Left panel
  document.getElementById('rItemCode').textContent = item.code;
  document.getElementById('rItemName').textContent = item.name;

  // Chips from name
  const dims = extractDims(item.name);
  const chips = dims.map(d=>`<span class="chip">${d.v}${d.u}</span>`);
  document.getElementById('rItemChips').innerHTML = chips.join('');

  // Nav info
  const done = queue.filter(x=>x.status==='done').length;
  document.getElementById('navInfo').textContent = `Stavka ${currentIdx+1} od ${queue.length}`;

  const best = item.allMatches[item.selectedIdx||0];
  const mc = document.getElementById('matchContent');

  if(!best){
    mc.innerHTML = `<div class="no-match">&#128262; Nema prijedloga iz cjenika za ovu stavku.<br><small>Probajte preformulirati opis.</small></div>`;
    document.getElementById('acceptedOv').classList.remove('show');
    return;
  }

  // Confidence ring SVG
  const R=30, C=188.5, offset = C - (C * best.confidence/100);
  const col = confColor(best.confidence);

  // Generate diff tags
  const qWords = [...tokenize(item.name)].filter(w=>w.length>3);
  const mWords = tokenize(best.naziv);
  const common = qWords.filter(w=>mWords.has(w));
  const diffTags = [
    ...common.slice(0,3).map(w=>`<span class="diff-tag tag-ok">&#10003; ${w}</span>`),
    ...(best.confidence<80?['<span class="diff-tag tag-warn">&#9888; Provjeri opis</span>']:[]),
    ...(extractDims(item.name).length&&dimScore(extractDims(item.name),extractDims(best.naziv))===1?
       ['<span class="diff-tag tag-ok">&#10003; Dimenzije se podudaraju</span>']:
       extractDims(item.name).length?['<span class="diff-tag tag-warn">&#9888; Dimenzije: provjeri</span>']:[])
  ].join('');

  mc.innerHTML = `
    <div class="match-top">
      <div class="conf-ring">
        <svg width="68" height="68" viewBox="0 0 68 68" style="transform:rotate(-90deg)">
          <circle cx="34" cy="34" r="${R}" fill="none" stroke-width="6" stroke="#e8e6e0"/>
          <circle cx="34" cy="34" r="${R}" fill="none" stroke-width="6" stroke="${col}"
            stroke-linecap="round" stroke-dasharray="${C}" stroke-dashoffset="${offset.toFixed(1)}"
            style="transition:stroke-dashoffset .6s"/>
        </svg>
        <div class="conf-pct" style="margin-top:-52px;position:relative;z-index:1">${best.confidence}%</div>
        <div class="conf-sub" style="margin-top:6px">sigurnost</div>
      </div>
      <div class="match-info">
        <div class="match-name">${best.naziv}</div>
        <div class="match-price">
          <span class="price-amt">${best.cijena.toLocaleString('hr',{minimumFractionDigits:2,maximumFractionDigits:2})}</span>
          <span class="price-unit">€ / ${best.jedinica}</span>
        </div>
        <div style="font-size:11px;color:var(--text2)">Sekcija: ${best.sekcija}</div>
      </div>
    </div>
    <div class="conf-bar-wrap">
      <div class="conf-bar" style="width:${best.confidence}%;background:${col}"></div>
    </div>
    <div class="conf-row">
      <span class="conf-label">Ocjena podudarnosti</span>
      <span style="font-size:11px;font-weight:600;color:${col}">${confLabel(best.confidence)}</span>
    </div>
    <div class="diff-tags">${diffTags}</div>
    <div class="action-row">
      <button class="btn accept" onclick="accept()">&#9989; Prihvati prijedlog</button>
      <button class="btn alt" id="altToggleBtn" onclick="toggleAlts()">
        &#128270; Odaberi drugu cijenu
        <span id="altChevron" style="margin-left:auto">&#8964;</span>
      </button>
    </div>
    <div class="alts-panel ${altsOpen?'open':''}" id="altsPanel">
      <div class="alts-header">Ostali prijedlozi — razvrstano po sigurnosti</div>
      ${item.allMatches.slice(1).map((a,i)=>`
        <div class="alt-item" onclick="selectAlt(${i+1})">
          <div class="alt-score" ${scoreClass(a.confidence)}>${a.confidence}%</div>
          <div class="alt-info">
            <div class="alt-name">${a.naziv}</div>
            <div style="font-size:10px;color:var(--text2);margin-top:2px">${a.sekcija}</div>
          </div>
          <div class="alt-price">${a.cijena.toLocaleString('hr',{minimumFractionDigits:2,maximumFractionDigits:2})} €</div>
        </div>`).join('') || '<div style="padding:8px;font-size:12px;color:var(--text2)">Nema više prijedloga.</div>'}
    </div>
  `;

  if(item.status==='done'){
    document.getElementById('acceptedOv').classList.add('show');
  } else {
    document.getElementById('acceptedOv').classList.remove('show');
  }
}

function renderProgress(){
  const done = queue.filter(x=>x.status!=='pending').length;
  const pct = queue.length?Math.round(done/queue.length*100):0;
  document.getElementById('progressFill').style.width = pct+'%';
  document.getElementById('progressText').textContent = `${done} / ${queue.length}`;
}

function updateStats(){
  const done = queue.filter(x=>x.status==='done').length;
  const pend = queue.filter(x=>x.status==='pending').length;
  const confs = queue.filter(x=>x.allMatches.length).map(x=>x.allMatches[x.selectedIdx||0]?.confidence||0);
  const avg = confs.length?Math.round(confs.reduce((a,b)=>a+b,0)/confs.length):0;
  document.getElementById('statTotal').textContent = queue.length;
  document.getElementById('statDone').textContent = done;
  document.getElementById('statPend').textContent = pend;
  document.getElementById('statAvgConf').textContent = avg?avg+'%':'—';
}

function accept(){
  if(currentIdx<0) return;
  queue[currentIdx].status='done';
  document.getElementById('acceptedOv').classList.add('show');
  setTimeout(()=>{
    const next = queue.findIndex((x,i)=>i>currentIdx&&x.status==='pending');
    if(next>=0) setActive(next);
    else renderQueue();
    updateStats();
    renderProgress();
  },700);
}

function skipItem(){
  if(currentIdx<0) return;
  queue[currentIdx].status='skip';
  const next = queue.findIndex((x,i)=>i>currentIdx&&x.status==='pending');
  if(next>=0) setActive(next);
  else{renderQueue();updateStats();renderProgress();}
}

function toggleAlts(){
  altsOpen=!altsOpen;
  const panel=document.getElementById('altsPanel');
  if(panel) panel.classList.toggle('open',altsOpen);
  const chev=document.getElementById('altChevron');
  if(chev) chev.style.transform=altsOpen?'rotate(180deg)':'';
}

function selectAlt(idx){
  if(currentIdx<0) return;
  queue[currentIdx].selectedIdx=idx;
  altsOpen=false;
  renderMatch();
  updateStats();
  showToast('Alternativni prijedlog odabran');
}

function navigate(dir){
  const next=currentIdx+dir;
  if(next>=0&&next<queue.length) setActive(next);
}

function showToast(msg){
  const t=document.getElementById('toast');
  t.textContent=msg;
  t.classList.add('show');
  setTimeout(()=>t.classList.remove('show'),2000);
}
